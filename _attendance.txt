function buildAttendanceTimeDropdowns_(){
  const ids=[
    'attendanceCheckInTime',
    'attendanceCheckOutTime'
  ];

  const options=['<option value="">Chọn mốc giờ</option>'];

  for(let h=0;h<24;h++){
    ['00','30'].forEach(function(m){
      const value=String(h).padStart(2,'0')+':'+m;
      options.push(
        '<option value="'+value+'">'+value+'</option>'
      );
    });
  }

  ids.forEach(function(id){
    const el=document.getElementById(id);
    if(el)el.innerHTML=options.join('');
  });
}

function getAttendanceTimeFromDropdown_(selectId){
  return document.getElementById(selectId)?.value||'';
}

function ensureAttendanceTimeOption_(selectId,time){
  const select=document.getElementById(selectId);
  if(!select||!time)return;

  const text=String(time).trim();
  const exists=[...select.options].some(function(o){
    return o.value===text;
  });

  if(!exists){
    const option=document.createElement('option');
    option.value=text;
    option.textContent=text+' (đã lưu)';
    select.appendChild(option);
  }
}

function setAttendanceTimeDropdown_(time,selectId){
  const text=String(time||'').trim();
  const select=document.getElementById(selectId);
  if(!select)return;

  if(!/^\d{2}:\d{2}$/.test(text)){
    select.value='';
    return;
  }

  ensureAttendanceTimeOption_(selectId,text);
  select.value=text;
}


function loadAttendanceOptions_(){
  google.script.run
    .withSuccessHandler(r=>{
      const select=document.getElementById('attendanceStore');
      if(!select)return;

      if(!r||!r.ok){
        select.innerHTML='<option value="">Không tải được chi nhánh</option>';
        return;
      }

      const stores=r.stores||[];

      if(!stores.length){
        select.innerHTML='<option value="">Chưa có chi nhánh</option>';
        return;
      }

      select.innerHTML=

        stores.map(s=>

      }

      const select=document.getElementById('attendanceStore');
      if(select)select.innerHTML='<option value="">Lỗi tải chi nhánh</option>';
    })

}


function getVietnamTodayKey_(){
  const now=new Date();

  const parts=new Intl.DateTimeFormat(
    'en-CA',
    {
      timeZone:'Asia/Ho_Chi_Minh',
      year:'numeric',
      month:'2-digit',
      day:'2-digit'
    }
  ).formatToParts(now);

  const map={};
  parts.forEach(p=>map[p.type]=p.value);

  return map.year+'-'+map.month+'-'+map.day;
}

function getVietnamMonthStartKey_(){
  const today=getVietnamTodayKey_();
  return today.slice(0,8)+'01';
}

function normalizeAttendanceTimeInput_(value){
  let text=String(value||'').trim();

  text=text.replace(/[^\d:]/g,'');

  if(/^\d{4}$/.test(text)){
    text=text.slice(0,2)+':'+text.slice(2);
  }

  const match=text.match(/^(\d{1,2})(?::(\d{0,2}))?$/);

  if(!match){
    return text;
  }

  let hour=match[1];
  let minute=match[2]||'';

  if(hour.length===1){
    hour='0'+hour;
  }

  if(minute.length===1){
    minute='0'+minute;
  }

  if(minute){
    return hour+':'+minute;
  }

  return hour+':';
}

function isValidAttendanceTime_(value){
  return /^(?:[01]\d|2[0-3]):[0-5]\d$/.test(
    String(value||'').trim()
  );
}

function attendanceTimeToMinutes_(value){
  const text=String(value||'').trim();
  const match=text.match(/^(\d{2}):(\d{2})$/);
  if(!match)return null;

  const h=Number(match[1]);
  const m=Number(match[2]);

  if(h<0||h>23||m<0||m>59)return null;

  return h*60+m;
}

function getAttendanceHistoryFilters_(){
  return {
    startDate:
      document.getElementById('attendanceHistoryStart')?.value||
      getVietnamMonthStartKey_(),

    endDate:
      document.getElementById('attendanceHistoryEnd')?.value||
      getVietnamTodayKey_()
  };
}

function clearAttendanceForm_(){
  const dateEl=document.getElementById('attendanceDate');
  const storeEl=document.getElementById('attendanceStore');
  const msg=document.getElementById('attendanceFormMsg');

  [
    'attendanceCheckInTime',
    'attendanceCheckOutTime'
  ].forEach(function(id){
    const el=document.getElementById(id);
    if(el)el.value='';
  });

  if(msg)msg.textContent='';
  window.__editingAttendanceId=null;

  const btn=document.getElementById('confirmAttendanceBtn');
  if(btn)btn.textContent='Xác nhận chấm công';

  if(storeEl&&storeEl.options.length>1){
    storeEl.selectedIndex=0;
  }

  if(dateEl){
    dateEl.value=getVietnamTodayKey_();
  }
}


function submitAttendanceForm_(){
  const btn=document.getElementById('confirmAttendanceBtn');
  const msg=document.getElementById('attendanceFormMsg');

  const data={
    id:window.__editingAttendanceId||'',
    date:document.getElementById('attendanceDate').value,
    store:document.getElementById('attendanceStore').value,
    checkIn:getAttendanceTimeFromDropdown_('attendanceCheckInTime'),
    checkOut:getAttendanceTimeFromDropdown_('attendanceCheckOutTime')
  };

  if(!data.date||!data.store||!data.checkIn||!data.checkOut){
    msg.textContent='Vui lòng chọn đầy đủ ngày, chi nhánh, giờ và phút.';
    msg.style.color='#c2414a';
    return;
  }

  if(!isValidAttendanceTime_(data.checkIn)||!isValidAttendanceTime_(data.checkOut)){
    msg.textContent='Giờ phải theo định dạng 24 giờ HH:mm.';
    msg.style.color='#c2414a';
    return;
  }

  const checkInMinutes=attendanceTimeToMinutes_(data.checkIn);
  const checkOutMinutes=attendanceTimeToMinutes_(data.checkOut);

  if(checkInMinutes==null||checkOutMinutes==null||checkOutMinutes<=checkInMinutes){
    msg.textContent='Giờ ra phải lớn hơn giờ vào.';
    msg.style.color='#c2414a';
    return;
  }

  btn.disabled=true;
  btn.textContent=data.id?'Đang cập nhật…':'Đang xác nhận…';

  const method=data.id?'updateAttendanceRecord':'createAttendanceRecord';

  google.script.run
    .withSuccessHandler(r=>{
      btn.disabled=false;

      if(r&&r.ok){
        msg.textContent=r.message||'Đã lưu chấm công.';
        msg.style.color='#0f9f75';
        window.__editingAttendanceId=null;
        btn.textContent='Xác nhận chấm công';
        loadAttendanceHistory_();
      }else{
        btn.textContent=data.id?'Cập nhật chấm công':'Xác nhận chấm công';
        msg.textContent=(r&&r.message)||'Không thể lưu chấm công.';
        msg.style.color='#c2414a';
      }
    })
    .withFailureHandler(err=>{
      btn.disabled=false;
      btn.textContent=data.id?'Cập nhật chấm công':'Xác nhận chấm công';
      msg.textContent='Không thể ghi chấm công: '+String(err&&err.message||'Lỗi kết nối từ máy chủ.');
      msg.style.color='#c2414a';
    })[method](currentSessionToken,data);
}


function startEditAttendance_(id){
  const rows=window.__attendanceHistoryItems||[];
  const item=rows.find(x=>String(x.id)===String(id));
  if(!item)return;

  if(!item.canEdit){
    alert('Bản chấm công này đã quá 24 giờ và không thể sửa.');
    return;
  }

  window.__editingAttendanceId=item.id;

  const dateEl=document.getElementById('attendanceDate');
  const storeEl=document.getElementById('attendanceStore');
  const btn=document.getElementById('confirmAttendanceBtn');
  const msg=document.getElementById('attendanceFormMsg');

  if(dateEl)dateEl.value=String(item.date||'').slice(0,10);

  if(storeEl){
    const exists=[...storeEl.options].some(o=>o.value===item.store);
    if(!exists&&item.store){
      const option=document.createElement('option');
      option.value=item.store;
      option.textContent=item.store;
      storeEl.appendChild(option);
    }
    storeEl.value=item.store||'';
  }

  setAttendanceTimeDropdown_(extractAttendanceTime_(item.checkIn),'attendanceCheckInTime');
  setAttendanceTimeDropdown_(extractAttendanceTime_(item.checkOut),'attendanceCheckOutTime');

  if(btn)btn.textContent='Cập nhật chấm công';

  if(msg){
    msg.textContent='Đang sửa bản chấm công. '+attendanceEditDeadlineText_(item.confirmedAt);
    msg.style.color='#a16207';
  }

  dateEl?.scrollIntoView({behavior:'smooth',block:'center'});
}


function loadAttendanceHistory_(){

    'attendanceHistoryFilterMsg'
  );

  if(filters.startDate && filters.endDate &&
     filters.startDate>filters.endDate){

    if(filterMsg){
      filterMsg.textContent=
        'Ngày bắt đầu phải nhỏ hơn hoặc bằng ngày kết thúc.';
      filterMsg.style.color='#c2414a';
    }

    return;
  }

  if(filterMsg){
    filterMsg.textContent=
      'Đang tải dữ liệu từ '+
      filters.startDate+
      ' đến '+
      filters.endDate+
      '…';
    filterMsg.style.color='#64748b';
  }

  google.script.run
    .withSuccessHandler(r=>{
      const box=document.getElementById('attendanceHistory');
      if(!box)return;

      if(!r||!r.ok){

        box.innerHTML=
          '<div class="employee-empty">'+
          ((r&&r.message)||'Không thể tải dữ liệu chấm công.')+
          '</div>';

          filterMsg.textContent=
            (r&&r.message)||'Không thể tải dữ liệu.';
          filterMsg.style.color='#c2414a';
        }

        return;

      if(filterMsg){
        filterMsg.textContent=
          'Đang hiển thị '+
          items.length+
          ' bản ghi • '+
          filters.startDate+
          ' → '+
          filters.endDate;

        filterMsg.style.color='#64748b';

      if(!items.length){

          'Không có dữ liệu chấm công trong khoảng đã chọn.'+

            '<thead><tr>'+

              '<th>Chi nhánh</th>'+
              '<th>Ca</th>'+
              '<th>Giờ vào</th>'+

              '<th>Giờ công</th>'+

              '<th>Thành tiền</th>'+
              '<th>Trạng thái</th>'+

              '<th>Thao tác</th>'+

                  editable

function updateAttendanceSummary_(items,filters){
  const list=Array.isArray(items)?items:[];

  const hours=list.reduce(
    (s,x)=>s+Number(x.hours||0),
    0
  );

  const amount=list.reduce(
    (s,x)=>s+Number(x.amount||0),
    0
  );

  const records=list.length;
  const average=records
    ?hours/records
    :0;

  const hoursEl=document.getElementById(
    'attendanceHoursToday'
  );

  const amountEl=document.getElementById(
    'attendanceAmountToday'
  );

  const recordsEl=document.getElementById(
    'attendanceRecordToday'
  );

  const averageEl=document.getElementById(
    'attendanceAverageHours'
  );

  const rangeEl=document.getElementById(
    'attendanceSummaryRange'
  );

  if(hoursEl){
    hoursEl.textContent=
      hours.toFixed(2).replace('.',',');
  }

  if(amountEl){
    amountEl.textContent=formatMoney_(amount);
  }

  if(recordsEl){
    recordsEl.textContent=String(records);
  }

  if(averageEl){
    averageEl.textContent=
      average.toFixed(2).replace('.',',');
  }

  if(rangeEl){
    rangeEl.textContent=
      'Theo khoảng '+
      String(filters&&filters.startDate||'')+
      ' → '+
      String(filters&&filters.endDate||'');
  }
}


function buildManagerAttendanceTimeOptions_(selectId,currentValue){
  const el=document.getElementById(selectId);
  if(!el)return;

  const current=String(currentValue||'').trim();
  const options=['<option value="">Chọn mốc giờ</option>'];

  for(let hour=0;hour<24;hour++){
    ['00','30'].forEach(function(minute){
      const value=String(hour).padStart(2,'0')+':'+minute;
      options.push('<option value="'+value+'">'+value+'</option>');
    });
  }

  if(
    current &&
    /^\d{2}:\d{2}$/.test(current) &&
    !options.some(x=>x.indexOf('value="'+current+'"')!==-1)
  ){
    options.push('<option value="'+current+'">'+current+' (đã lưu)</option>');
  }

  el.innerHTML=options.join('');
  if(current)el.value=current;
}

function getVietnamDateInputToday_(){

    year:'numeric',
    month:'2-digit',
    day:'2-digit'
  }).format(new Date());
}

function getVietnamMonthStartInput_(){
  const today=getVietnamDateInputToday_();
  return today.slice(0,8)+'01';
}
