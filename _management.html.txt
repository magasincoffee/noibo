function closeManagerAttendanceModal_(){
  document.getElementById('managerAttendanceModal')?.classList.remove('active');
}

function openManagerAttendanceEdit_(item){
  const modal=document.getElementById('managerAttendanceModal');
  const id=document.getElementById('managerEditAttendanceId');
  const date=document.getElementById('managerEditDate');
  const store=document.getElementById('managerEditStore');
  const reason=document.getElementById('managerEditReason');
  const msg=document.getElementById('managerEditMsg');

  if(!modal||!id||!date||!store)return;

  id.value=item.id||'';

  let dateKey=String(item.date||'').trim();
  if(/^\d{2}\/\d{2}\/\d{4}$/.test(dateKey)){
    dateKey=dateKey.replace(/^(\d{2})\/(\d{2})\/(\d{4})$/,'$3-$2-$1');
  }
  date.value=/^\d{4}-\d{2}-\d{2}$/.test(dateKey)?dateKey:'';

  const selectedStore=String(item.store||'');
  const managerStore=document.getElementById('managerAttendanceStore');
  const stores=managerStore
    ?Array.from(managerStore.options).filter(o=>o.value).map(o=>o.value)
    :[];

  store.innerHTML=stores.map(s=>'<option value="'+escapeClientHtml(s)+'">'+escapeClientHtml(s)+'</option>').join('');

  if(selectedStore&&!Array.from(store.options).some(o=>o.value===selectedStore)){
    const opt=document.createElement('option');
    opt.value=selectedStore;
    opt.textContent=selectedStore;
    store.appendChild(opt);
  }
  store.value=selectedStore;

  buildManagerAttendanceTimeOptions_('managerEditCheckIn',String(item.checkIn||''));
  buildManagerAttendanceTimeOptions_('managerEditCheckOut',String(item.checkOut||''));

  if(reason)reason.value='';
  if(msg){msg.textContent='';msg.style.color='';}

  modal.classList.add('active');
}

function submitManagerAttendanceEdit_(){
  const btn=document.getElementById('managerEditSave');
  const msg=document.getElementById('managerEditMsg');

  const data={
    id:document.getElementById('managerEditAttendanceId').value,
    date:document.getElementById('managerEditDate').value,
    store:document.getElementById('managerEditStore').value,
    checkIn:document.getElementById('managerEditCheckIn').value,
    checkOut:document.getElementById('managerEditCheckOut').value,
    reason:document.getElementById('managerEditReason').value
  };

  if(!data.id||!data.date||!data.store||!data.checkIn||!data.checkOut){
    if(msg){msg.textContent='Vui lòng nhập đủ thông tin.';msg.style.color='#c2414a';}
    return;
  }

  btn.disabled=true;
  btn.textContent='Đang lưu…';

  google.script.run
    .withSuccessHandler(r=>{
      btn.disabled=false;
      btn.textContent='Lưu thay đổi';

      if(msg){
        msg.textContent=r?.message||'';
        msg.style.color=r?.ok?'#0f9f75':'#c2414a';
      }

      if(r?.ok){
        setTimeout(function(){
          closeManagerAttendanceModal_();
          loadAttendanceManagement_();
        },350);
      }
    })
    .withFailureHandler(err=>{
      btn.disabled=false;
      btn.textContent='Lưu thay đổi';
      if(msg){
        msg.textContent='Không thể cập nhật: '+String(err?.message||'Lỗi kết nối.');
        msg.style.color='#c2414a';
      }
    })
    .managerUpdateAttendanceRecord(currentSessionToken,data);
}

function deleteManagerAttendance_(item){
  const reason=window.prompt('Nhập lý do xóa bản chấm công này:','');

  if(reason===null)return;

  if(!confirm(
    'Xóa bản chấm công ngày '+String(item.date||'')+
    ' của '+String(item.name||item.username||'')+
    ' tại '+String(item.store||'')+
    ' từ '+String(item.checkIn||'')+
    ' đến '+String(item.checkOut||'')+
    '?'
  )) return;

  google.script.run
    .withSuccessHandler(r=>{
      alert(r?.message||'');
      if(r?.ok)loadAttendanceManagement_();
    })
    .withFailureHandler(err=>{
      alert('Không thể xóa: '+String(err?.message||'Lỗi kết nối.'));
    })
    .managerDeleteAttendanceRecord(
      currentSessionToken,
      {id:item.id,reason:reason}
    );
}

function loadAttendanceManagement_(){
  const start=document.getElementById('managerAttendanceStart')?.value||'';
  const end=document.getElementById('managerAttendanceEnd')?.value||'';
  const store=document.getElementById('managerAttendanceStore')?.value||'';
  const msg=document.getElementById('managerAttendanceMsg');

  if(msg){msg.textContent='Đang tải báo cáo…';msg.style.color='#64748b';}

  google.script.run
    .withSuccessHandler(r=>{
      if(!r?.ok){
        if(msg){msg.textContent=r?.message||'Không thể tải báo cáo chấm công.';msg.style.color='#c2414a';}
        return;
      }

      const sel=document.getElementById('managerAttendanceStore');
      if(sel){
        const visibleStores=Array.isArray(r.stores)?r.stores:[];
        const previousValue=String(store||'');
        sel.innerHTML='<option value="">Tất cả chi nhánh</option>'+
          visibleStores.map(s=>{
            const safe=escapeClientHtml(String(s));
            return '<option value="'+safe+'">'+safe+'</option>';
          }).join('');

        if(previousValue && visibleStores.some(s=>String(s)===previousValue)){
          sel.value=previousValue;
        }else{
          sel.value='';
        }
      }

      document.getElementById('managerTotalHours').textContent=Number(r.totals.hours||0).toFixed(2).replace('.',',');
      document.getElementById('managerTotalAmount').textContent=formatMoney_(r.totals.amount);
      document.getElementById('managerEmployeeCount').textContent=r.totals.employees;
      document.getElementById('managerRecordCount').textContent=r.totals.records;

      const storeMap={};
      (r.daily||[]).forEach(x=>{
        const key=String(x.store||'').toUpperCase();
        if(!storeMap[key]){
          storeMap[key]={store:x.store,hours:0,amount:0,records:0,employees:0};
        }
        storeMap[key].hours+=Number(x.hours||0);
        storeMap[key].amount+=Number(x.amount||0);
        storeMap[key].records+=Number(x.records||0);
        storeMap[key].employees+=Number(x.employeeCount||0);
      });

      const storeItems=Object.keys(storeMap).map(k=>storeMap[k]).sort((a,b)=>b.hours-a.hours);
      const maxHours=storeItems.length?Math.max(...storeItems.map(x=>x.hours)):0;
      const storeBox=document.getElementById('managerStoreSummary');

      if(!storeItems.length){
        storeBox.innerHTML='<div class="employee-empty">Không có dữ liệu trong khoảng đã chọn.</div>';
      }else{
        storeBox.innerHTML='<div class="attendance-store-grid">'+
          storeItems.map(x=>{
            const pct=maxHours?Math.max(4,Math.min(100,x.hours/maxHours*100)):0;
            return '<div class="attendance-store-card">'+
              '<strong>'+escapeClientHtml(x.store)+'</strong>'+
              '<div class="store-hours">'+Number(x.hours||0).toFixed(2)+' giờ</div>'+
              '<div class="store-meta"><span>'+formatMoney_(x.amount)+'</span><span>'+x.records+' lượt</span></div>'+
              '<div class="attendance-mini-bar"><span style="width:'+pct.toFixed(1)+'%"></span></div>'+
            '</div>';
          }).join('')+
        '</div>';
      }

      const dailyBox=document.getElementById('managerDailySummary');
      dailyBox.innerHTML=r.daily&&r.daily.length
        ?'<div class="attendance-table-wrap"><table class="attendance-table"><thead><tr><th>Ngày</th><th>Chi nhánh</th><th>NV</th><th>Lượt</th><th>Tổng giờ</th><th>Tổng tiền</th></tr></thead><tbody>'+
          r.daily.map(x=>
            '<tr><td>'+escapeClientHtml(x.date)+'</td><td>'+escapeClientHtml(x.store)+'</td><td>'+Number(x.employeeCount||0)+'</td><td>'+Number(x.records||0)+'</td><td>'+Number(x.hours||0).toFixed(2)+'</td><td class="attendance-money">'+formatMoney_(x.amount)+'</td></tr>'
          ).join('')+
          '</tbody></table></div>'
        :'<div class="employee-empty">Không có dữ liệu.</div>';

      const detailBox=document.getElementById('managerAttendanceDetail');
      detailBox.innerHTML=r.details&&r.details.length
        ?'<div class="attendance-table-wrap"><table class="attendance-table"><thead><tr><th>Ngày</th><th>Chi nhánh</th><th>Nhân viên</th><th>Vào</th><th>Ra</th><th>Giờ công</th><th>Bậc</th><th>Thành tiền</th><th>Thao tác</th></tr></thead><tbody>'+
          r.details.map(x=>
            '<tr><td>'+escapeClientHtml(x.date)+'</td><td>'+escapeClientHtml(x.store)+'</td><td>'+escapeClientHtml(x.name||x.username)+'</td><td>'+formatAttendanceTime_(x.checkIn)+'</td><td>'+formatAttendanceTime_(x.checkOut)+'</td><td>'+Number(x.hours||0).toFixed(2)+'</td><td>'+escapeClientHtml(x.grade||'—')+'</td><td class="attendance-money">'+formatMoney_(x.amount)+'</td><td><div class="attendance-manager-actions"><button type="button" class="attendance-manager-edit" data-manager-att-edit="'+escapeClientHtml(x.id)+'">Sửa</button><button type="button" class="attendance-manager-delete" data-manager-att-del="'+escapeClientHtml(x.id)+'">Xóa</button></div></td></tr>'
          ).join('')+
          '</tbody></table></div>'
        :'<div class="employee-empty">Không có dữ liệu.</div>';

      window.__managerAttendanceDetails=Array.isArray(r.details)?r.details:[];

      detailBox.querySelectorAll('[data-manager-att-edit]').forEach(btn=>{
        btn.addEventListener('click',function(){
          const item=window.__managerAttendanceDetails.find(x=>String(x.id)===String(btn.dataset.managerAttEdit));
          if(item)openManagerAttendanceEdit_(item);
        });
      });

      detailBox.querySelectorAll('[data-manager-att-del]').forEach(btn=>{
        btn.addEventListener('click',function(){
          const item=window.__managerAttendanceDetails.find(x=>String(x.id)===String(btn.dataset.managerAttDel));
          if(item)deleteManagerAttendance_(item);
        });
      });

      if(msg){
        if(Number(r.totals.records||0)>0){
          const scope=String((r.diagnostics&&r.diagnostics.accessScope)||'').trim();
          msg.textContent='Đang hiển thị '+r.totals.records+' bản ghi • '+start+' → '+end+
            (scope?' • Phạm vi: '+scope:'');
          msg.style.color='#64748b';
        }else{
          const d=r.diagnostics||{};
          const scope=String(d.accessScope||'').trim();
          const allStores=(d.allStores||[]).join(', ');
          const dateStores=(d.storesInDateRange||[]).join(', ');

          if(Number(d.matchedDateRows||0)===0){
            msg.textContent='Nguồn: '+String(d.spreadsheetName||'')+
              ' | Tab: '+String(d.sheetName||'Chấm công')+
              ' | Dòng: '+Number(d.sheetLastRow||0)+
              ' | Có '+Number(d.rawDataRows||0)+' bản ghi nhưng không có bản ghi nào nằm trong khoảng '+start+' → '+end+'.';
          }else if(Number(d.matchedStoreRows||0)===0){
            msg.textContent='Có '+Number(d.matchedDateRows||0)+' bản ghi trong khoảng ngày, nhưng không khớp chi nhánh đang chọn.';
          }else if(Number(d.matchedScopeRows||0)===0){
            msg.textContent=scope
              ? 'Có dữ liệu trong khoảng ngày nhưng không nằm trong Phạm vi truy cập ['+scope+']. Cửa hàng có dữ liệu: '+dateStores+'.'
              : 'Tài khoản quản lý chưa có Phạm vi truy cập trong Sheet Người dùng (cột P). Cửa hàng có dữ liệu: '+dateStores+'.';
          }else{
            msg.textContent='Không có bản ghi chấm công hợp lệ trong khoảng đã chọn.';
          }

          if(allStores && !dateStores){
            msg.textContent+=' Tổng Sheet hiện có: '+allStores+'.';
          }

          msg.style.color='#c2414a';
        }
      }
    })
    .withFailureHandler(err=>{
      if(msg){msg.textContent='Không thể tải báo cáo: '+String(err?.message||'Lỗi kết nối.');msg.style.color='#c2414a';}
    })
    .getAttendanceManagementV33(currentSessionToken,{startDate:start,endDate:end,store:store});
}

  if(page==='inventory' && role==='STAFF'){
    document.querySelectorAll('.menu-item').forEach(btn=>{
      btn.classList.toggle('active',btn.dataset.page==='inventory');
    });
  }

  _showPageRoleTasks(page);
};

document.addEventListener('click',function(e){
  const sidebarTask=e.target.closest('[data-staff-task-sidebar]');
  if(sidebarTask){
    const role=normalizedClientRole(currentUser&&currentUser.role);
    if(role!=='STAFF') return;
    document.querySelectorAll('.staff-task-menu-item').forEach(x=>{
      x.classList.toggle('active',x===sidebarTask);
    });
    renderStaffTaskPlaceholder_(sidebarTask.dataset.staffTaskSidebar);
    return;
  }

  const taskButton=e.target.closest('[data-staff-task]');
  if(taskButton){
    renderStaffTaskPlaceholder_(taskButton.dataset.staffTask);
    return;
  }

  if(e.target.closest('#backToStaffTasks')){
    document.querySelectorAll('.staff-task-menu-item').forEach(x=>{
      x.classList.remove('active');
    });
    showPage('inventory');
  }
});
